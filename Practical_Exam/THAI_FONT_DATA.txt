// ฟังก์ชันระบุประเภทตัวอักษรไทย
inline ThaiCharType getThaiCharType(uint8_t c) {
    if ((c >= 0xD4 && c <= 0xD7) || c == 0xD1 || c == 0xE7) return THAI_TYPE_UPPER;
    if (c >= 0xE8 && c <= 0xEC) return THAI_TYPE_TONE;
    if (c >= 0xD8 && c <= 0xDA) return THAI_TYPE_LOWER;
    return THAI_TYPE_BASE;
}
