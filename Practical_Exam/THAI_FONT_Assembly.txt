; ------------------------------------------------------------------------------
; void thai_get_glyph_asm(uint8_t tis_char, uint8_t* dest_buffer)
; Parameters:
;   r24     = TIS-620 character code (0xA1 - 0xFB)
;   r23:r22 = Pointer to 8-byte destination array in SRAM
; Description:
;   ดึงบิตแมป 8 ไบต์ของตัวอักษรไทยจาก Flash เข้าสู่ SRAM ผ่านคำสั่ง `lpm`
; ------------------------------------------------------------------------------
    .global thai_get_glyph_asm
thai_get_glyph_asm:
    cpi   r24, 0xA1            ; เช็คว่าต่ำกว่า ก หรือไม่
    brcs  .L_fill_blank
    cpi   r24, 0xFC            ; เช็คว่าเกิน 0xFB หรือไม่
    brcc  .L_fill_blank

    ; Offset = (tis_char - 0xA1) * 8
    subi  r24, 0xA1
    clr   r25
    lsl   r24                  ; * 2
    rol   r25
    lsl   r24                  ; * 4
    rol   r25
    lsl   r24                  ; * 8
    rol   r25

    ; Z = thai_font_table + Offset
    ldi   r30, lo8(thai_font_table)
    ldi   r31, hi8(thai_font_table)
    add   r30, r24
    adc   r31, r25

    ; X = dest_buffer in SRAM
    movw  r26, r22

    ; อ่าน Flash 8 ไบต์ด้วย lpm เข้า SRAM
    ldi   r21, 8
.L_copy_loop:
    lpm   r0, Z+
    st    X+, r0
    dec   r21
    brne  .L_copy_loop
    ret

.L_fill_blank:
    movw  r26, r22
    ldi   r21, 8
    clr   r0
.L_blank_loop:
    st    X+, r0
    dec   r21
    brne  .L_blank_loop
    ret


; ------------------------------------------------------------------------------
; uint8_t thai_classify_char_asm(uint8_t tis_char)
; Parameters:
;   r24 = TIS-620 character code
; Returns:
;   r24 = 0 (Base), 1 (Upper Vowel), 2 (Tone Mark), 3 (Lower Vowel)
; ------------------------------------------------------------------------------
    .global thai_classify_char_asm
thai_classify_char_asm:
    ; ตรวจสอบสระล่าง: 0xD8 (ุ), 0xD9 (ู), 0xDA (ฺ)
    cpi   r24, 0xD8
    brcs  .L_chk_upper
    cpi   r24, 0xDB
    brcc  .L_chk_tone
    ldi   r24, 3               ; Lower
    ret

.L_chk_tone:
    ; ตรวจสอบวรรณยุกต์: 0xE8 (่) ถึง 0xEC (์)
    cpi   r24, 0xE8
    brcs  .L_chk_upper
    cpi   r24, 0xED
    brcc  .L_ret_base
    ldi   r24, 2               ; Tone
    ret

.L_chk_upper:
    ; ตรวจสอบสระบน: 0xD4 (ิ) ถึง 0xD7 (ื), 0xD1 (ั), 0xE7 (็)
    cpi   r24, 0xD1
    breq  .L_ret_upper
    cpi   r24, 0xE7
    breq  .L_ret_upper
    cpi   r24, 0xD4
    brcs  .L_ret_base
    cpi   r24, 0xD8
    brcc  .L_ret_base

.L_ret_upper:
    ldi   r24, 1               ; Upper
    ret

.L_ret_base:
    clr   r24                  ; Base
    ret
